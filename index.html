-- Stock Manager: run this whole file once in Supabase > SQL Editor create table public.items ( id uuid primary key default gen_random_uuid(), sku text not null unique, name text not null, category text not null, location text not null default '', qty integer not null default 0 check (qty >= 0), reorder_level integer not null default 5 check (reorder_level >= 0), price numeric(12,2) not null default 0 check (price >= 0), created_at timestamptz not null default now() ); create table public.movements ( id bigint generated always as identity primary key, item_id uuid references public.items(id) on delete set null, item_name text not null, delta integer not null, note text, created_by uuid default auth.uid(), created_at timestamptz not null default now() ); create index movements_created_idx on public.movements (created_at desc); -- Row Level Security: only signed-in users can touch data alter table public.items enable row level security; alter table public.movements enable row level security; revoke all on public.items, public.movements from anon; create policy "signed-in users manage items" on public.items for all to authenticated using (true) with check (true); create policy "signed-in users read movements" on public.movements for select to authenticated using (true); -- Every stock change is logged automatically (clients cannot write movements directly) create or replace function public.log_item_change() returns trigger language plpgsql security definer set search_path = public as $$ declare n text := nullif(current_setting('app.note', true), ''); begin if tg_op = 'INSERT' then insert into movements (item_id, item_name, delta, note) values (new.id, new.name, new.qty, 'New item added'); elsif tg_op = 'UPDATE' then if new.qty <> old.qty then insert into movements (item_id, item_name, delta, note) values (new.id, new.name, new.qty - old.qty, coalesce(n, 'Quantity edited')); end if; else insert into movements (item_id, item_name, delta, note) values (null, old.name, 0, 'Item deleted'); end if; return coalesce(new, old); end $$; create trigger items_log after insert or update or delete on public.items for each row execute function public.log_item_change(); -- Atomic +/- adjustment (safe when several people edit at once) create or replace function public.adjust_stock(p_item uuid, p_delta integer, p_note text default null) returns void language plpgsql as $$ begin perform set_config('app.note', coalesce(p_note, ''), true); update public.items set qty = qty + p_delta where id = p_item; if not found then raise exception 'Item not found'; end if; end $$; revoke execute on function public.adjust_stock(uuid, integer, text) from public, anon;
